# Plan 6: kotlinx-rpc Layer Wiring — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** frontend が呼び出せる kotlinx-rpc API の枠組み(domain 集約の @Serializable 化、shared/rpc に service interface、backend に Service 実装・認証スタブ・StatusPages・Routing・DI 配線)を整える。本格認証と結合テストは別 Plan。

**Architecture:** 集約と関連 enum / 内部コレクション型を全て `@Serializable` 化し、RPC 経由で domain 型を直接やり取りする。中間 DTO や mapper は作らない。Service 実装は Repository を直接保持して「Principal → User 集約取得 → ID → 集約取得 → Handler 呼び出し」を行う。認証は Bearer token = UserId 文字列のスタブ Plugin。例外は StatusPages で HTTP ステータスへ変換。

**Tech Stack:** Kotlin Multiplatform, kotlinx.serialization, kotlinx-rpc 0.10.2 (krpc), Ktor Server 3.x, Ktor DI plugin, Ktor Authentication, Ktor StatusPages, Kotest, MockK

**Spec:** `docs/superpowers/specs/2026-05-25-rpc-layer-design.md`

```kotlin
package net.brightroom.mindstock.domain.model.stock

import kotlinx.serialization.Serializable
import net.brightroom.mindstock.domain.model.product.Product
import net.brightroom.mindstock.domain.model.stock.movement.StockMovements

@Serializable
data class Stock(
    val product: Product,
    val movements: StockMovements,
) {
    fun currentQuantity(): Int = movements.netQuantity()
    fun needsReplenishment(): Boolean {
        val minimum = product.minimumStock?.let { it() } ?: return false
        return currentQuantity() < minimum
    }
    fun shortage(): Int {
        val minimum = product.minimumStock?.let { it() } ?: 0
        return (minimum - currentQuantity()).coerceAtLeast(0)
    }
}
```

```kotlin
@OptIn(ExperimentalUuidApi::class)
class SerializationRoundTripTest : StringSpec({

    val json = Json { encodeDefaults = true }

    fun <T> roundTrip(value: T, serializer: kotlinx.serialization.KSerializer<T>): T =
        json.decodeFromString(serializer, json.encodeToString(serializer, value))

    val user = User(
        id = UserId(Uuid.parse("00000000-0000-0000-0000-000000000001")),
        authIdentity = AuthIdentity(
            provider = AuthProvider.GOOGLE,
            subject = AuthSubject("subject-1"),
        ),
        displayName = DisplayName("Alice"),
    )

        id = CatalogItemId(Uuid.parse("00000000-0000-0000-0000-000000000002")),
        name = CatalogItemName("Milk"),
        unit = CatalogItemUnit("L"),
    )

    val product = Product(
        id = ProductId(Uuid.parse("00000000-0000-0000-0000-000000000003")),
        catalogItem = catalogItem,
        minimumStock = MinimumStock(2),
        archived = false,
    )

    val replenishment: StockMovement = Replenishment(
        product = product,
        quantity = Quantity(5),
        occurredAt = OccurredAt(Instant.parse("2026-05-25T10:00:00Z")),
        actor = user,
        note = Note(""),
    )

    val consumption: StockMovement = Consumption(
        product = product,
        quantity = Quantity(1),
        occurredAt = OccurredAt(Instant.parse("2026-05-25T11:00:00Z")),
        actor = user,
        note = Note("breakfast"),
    )

    "User round-trip" {
        roundTrip(user, User.serializer()) shouldBe user
    }

    "CatalogItem round-trip" {
        roundTrip(catalogItem, CatalogItem.serializer()) shouldBe catalogItem
    }

    "Product round-trip" {
        roundTrip(product, Product.serializer()) shouldBe product
    }

    "Replenishment as StockMovement round-trip (polymorphic)" {
        roundTrip(replenishment, StockMovement.serializer()) shouldBe replenishment
    }

    "Consumption as StockMovement round-trip (polymorphic)" {
        roundTrip(consumption, StockMovement.serializer()) shouldBe consumption
    }

    "Stock round-trip" {
        val stock = Stock(
            product = product,
            movements = StockMovements(listOf(replenishment, consumption)),
        )
        val restored = roundTrip(stock, Stock.serializer())
        restored.product shouldBe product
        restored.movements.list shouldBe listOf(replenishment, consumption)
        restored.currentQuantity() shouldBe 4
    }

    "Household round-trip" {
        val household = Household(
            id = HouseholdId(Uuid.parse("00000000-0000-0000-0000-000000000004")),
            members = HouseholdMembers(
                list = listOf(HouseholdMember(user, HouseholdMemberRole.OWNER)),
            ),
        )
        roundTrip(household, Household.serializer()) shouldBe household
    }
})
```

```kotlin
import net.brightroom.mindstock.domain.model.stock.movement.StockMovement
import net.brightroom.mindstock.domain.repository.household.HouseholdRepository
import net.brightroom.mindstock.domain.repository.product.ProductRepository
import net.brightroom.mindstock.domain.repository.user.UserRepository
import net.brightroom.mindstock.presentation.rpc.StockRpcService

class StockRpcServiceImpl(
    private val get: GetStockHandler,
    private val list: ListStocksHandler,
    private val movementHistory: GetMovementHistoryHandler,
    private val replenish: ReplenishStockHandler,
    private val consume: ConsumeStockHandler,
    private val productRepository: ProductRepository,
    private val householdRepository: HouseholdRepository,
    private val userRepository: UserRepository,
) : StockRpcService {
    override suspend fun get(productId: ProductId): Stock {
        currentCall().actor(userRepository)
        val product = productRepository.findById(productId)
            ?: throw NotFoundException("product not found: $productId")
        return get.handle(product)
    }

    override suspend fun list(householdId: HouseholdId): List<Stock> {
        currentCall().actor(userRepository)
        val household = householdRepository.findById(householdId)
            ?: throw NotFoundException("household not found: $householdId")
        return list.handle(household)
    }

    override suspend fun movementHistory(productId: ProductId, limit: Int): List<StockMovement> {
        currentCall().actor(userRepository)
        val product = productRepository.findById(productId)
            ?: throw NotFoundException("product not found: $productId")
        return movementHistory.handle(product, limit).asList()
    }

    override suspend fun replenish(productId: ProductId, qty: Quantity, occurredAt: OccurredAt, note: Note): Replenishment {
        val actor = currentCall().actor(userRepository)
        val product = productRepository.findById(productId)
            ?: throw NotFoundException("product not found: $productId")
        return replenish.handle(product, qty, occurredAt, actor, note)
    }

    override suspend fun consume(productId: ProductId, qty: Quantity, occurredAt: OccurredAt, note: Note): Consumption {
        val actor = currentCall().actor(userRepository)
        val product = productRepository.findById(productId)
            ?: throw NotFoundException("product not found: $productId")
        return consume.handle(product, qty, occurredAt, actor, note)
    }
}
```

```kotlin
import io.ktor.serialization.kotlinx.json.jsonIo
import io.ktor.server.application.Application
import io.ktor.server.application.install
import io.ktor.server.auth.authenticate
import io.ktor.server.plugins.contentnegotiation.ContentNegotiation
import io.ktor.server.plugins.di.annotations.Property
import io.ktor.server.plugins.di.dependencies
import io.ktor.server.routing.routing
import kotlinx.rpc.krpc.ktor.server.rpc
import kotlinx.rpc.krpc.ktor.server.Krpc
import kotlinx.rpc.krpc.serialization.json.json
import kotlinx.serialization.ExperimentalSerializationApi
import net.brightroom.mindstock.configuration.Environment
import net.brightroom.mindstock.extensions.kotlinx.serialization.CustomJson
import net.brightroom.mindstock.presentation.rpc.CatalogRpcService
import net.brightroom.mindstock.presentation.rpc.HouseholdRpcService
import net.brightroom.mindstock.presentation.rpc.ProductRpcService
import net.brightroom.mindstock.presentation.rpc.StockRpcService
import net.brightroom.mindstock.presentation.rpc.UserPublicRpcService
import net.brightroom.mindstock.presentation.rpc.UserRpcService

fun Application.routingConfigure(
    @Property("ktor.environment") environment: Environment,
) {
    install(ContentNegotiation) {
        jsonIo(CustomJson)
    }
    install(Krpc) {
        serialization { json(CustomJson) }
    }

    val deps = dependencies

    routing {
        // 認証不要
        rpc("/api/v1/user/public") {
            registerService<UserPublicRpcService> { deps.resolve() }
        }
        // 認証必要
        authenticate("user") {
            rpc("/api/v1/user")      { registerService<UserRpcService>      { deps.resolve() } }
            rpc("/api/v1/household") { registerService<HouseholdRpcService> { deps.resolve() } }
            rpc("/api/v1/catalog")   { registerService<CatalogRpcService>   { deps.resolve() } }
            rpc("/api/v1/product")   { registerService<ProductRpcService>   { deps.resolve() } }
            rpc("/api/v1/stock")     { registerService<StockRpcService>     { deps.resolve() } }
        }
    }
}
```
